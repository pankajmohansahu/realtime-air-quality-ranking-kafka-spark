================================================================================
  WINDOW 3 — KAFKA SSH TAB 2
  Purpose: Configure server.properties + START KAFKA BROKER (keep running)
  Open new Cloud Shell tab → SSH into Kafka VM
================================================================================

--- Step 1: SSH into Kafka VM (new tab) ---
gcloud compute ssh kafka-vm-oppe2 --zone=us-central1-a

--- Step 2: Auto-detect external IP and configure Kafka ---
cd ~/kafka
KAFKA_EXTERNAL_IP=$(curl -s ifconfig.me)
echo "External IP detected: $KAFKA_EXTERNAL_IP"

--- Step 3: Update listeners in server.properties ---
grep -q "^listeners=" config/server.properties \
    && sed -i 's|^listeners=.*|listeners=PLAINTEXT://0.0.0.0:9092|' config/server.properties \
    || echo "listeners=PLAINTEXT://0.0.0.0:9092" >> config/server.properties

--- Step 4: Update advertised.listeners in server.properties ---
grep -q "^advertised.listeners=" config/server.properties \
    && sed -i "s|^advertised.listeners=.*|advertised.listeners=PLAINTEXT://$KAFKA_EXTERNAL_IP:9092|" config/server.properties \
    || echo "advertised.listeners=PLAINTEXT://$KAFKA_EXTERNAL_IP:9092" >> config/server.properties

--- Step 5: Verify config was written correctly ---
grep "listeners" config/server.properties

--- Expected output ---
listeners=PLAINTEXT://0.0.0.0:9092
advertised.listeners=PLAINTEXT://34.56.39.5:9092

--- Step 6: Start Kafka Broker (KEEP THIS TERMINAL RUNNING — do not close) ---
bin/kafka-server-start.sh config/server.properties

>>> Wait until you see: "started (kafka.server.KafkaServer)"
>>> Then open KAFKA SSH TAB 3 to create the topic

  Purpose: Create Kafka topic (run once, then you can close this tab)

--- Step 1: SSH into Kafka VM (new tab) ---

--- Step 2: Create the topic ---
cd ~/kafka

    --topic air-quality-topic \
    --bootstrap-server localhost:9092 \

    --replication-factor 1
